41class Conference(ObjectIDMixin, BaseModel): 

42 #: Determines the email address for submission and the GakuNin RDM url 

43 # Example: If endpoint is spsp2014, then submission email will be 

44 # spsp2014-talk@osf.io or spsp2014-poster@osf.io and the GakuNin RDM url will 

45 # be osf.io/view/spsp2014 

46 endpoint = models.CharField(max_length=255, unique=True, db_index=True) 

47 #: Full name, e.g. "SPSP 2014" 

48 name = models.CharField(max_length=255) 

49 info_url = models.URLField(blank=True) 

50 logo_url = models.URLField(blank=True) 

51 location = models.CharField(max_length=2048, null=True, blank=True) 

52 start_date = NonNaiveDateTimeField(blank=True, null=True) 

53 end_date = NonNaiveDateTimeField(blank=True, null=True) 

54 is_meeting = models.BooleanField(default=True) 

55 active = models.BooleanField() 

56 admins = models.ManyToManyField('OSFUser') 

57 # Temporary field on conference model to link Conferences and AbstractNodes 

58 submissions = models.ManyToManyField('AbstractNode', related_name='conferences') 

59 # Whether to make submitted projects public 

60 public_projects = models.BooleanField(default=True) 

61 poster = models.BooleanField(default=True) 

62 talk = models.BooleanField(default=True) 

63 # field_names are used to customize the text on the conference page, the categories 

64 # of submissions, and the email adress to send material to. 

65 field_names = DateTimeAwareJSONField(default=get_default_field_names)
