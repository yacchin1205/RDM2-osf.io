14class scope(namedtuple('scope', ['parts_', 'description', 'is_public'])): 

15 """ Patch to add `ALWAYS_PUBLIC` scope to every selectable scope, 

16 ensuring that public endpoints are accessible with any token.

18 @property 

19 def parts(self): 

20 return frozenset((CoreScopes.ALWAYS_PUBLIC, )).union(self.parts_)

374def normalize_scopes(scopes): 

375 """ 

376 Given a list of public-facing scope names from a CAS token, return the list of internal scopes 

377 

378 This is useful for converting a single broad scope name (from CAS) into the small constituent parts 

379 (as used by views) 

380 

381 :param list scopes: a list public facing scopes 

382 """ 

383 all_scopes = set() 

384 for sc in scopes: 

385 try: 

386 scope_tuple = public_scopes[sc] 

387 all_scopes |= scope_tuple.parts 

388 except KeyError: 

389 pass 

390 return all_scopes
