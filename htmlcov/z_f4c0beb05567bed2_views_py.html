257def resolve_guid(guid, suffix=None): 

258 """Load GUID by primary key, look up the corresponding view function in the 

259 routing table, and return the return value of the view function without 

260 changing the URL. 

261 

262 :param str guid: GUID primary key 

263 :param str suffix: Remainder of URL after the GUID 

264 :return: Return value of proxied view function 

265 """ 

266 try: 

267 # Look up 

268 guid_object = Guid.load(guid) 

269 except KeyError as e: 

270 if e.message == 'osfstorageguidfile': # Used when an old detached OsfStorageGuidFile object is accessed 

271 raise HTTPError(http_status.HTTP_404_NOT_FOUND) 

272 else: 

273 raise e 

274 if guid_object: 

275 # verify that the object implements a GuidStoredObject-like interface. If a model 

276 # was once GuidStoredObject-like but that relationship has changed, it's 

277 # possible to have referents that are instances of classes that don't 

278 # have a deep_url attribute or otherwise don't behave as 

279 # expected. 

280 if not hasattr(guid_object.referent, 'deep_url'): 

281 sentry.log_message( 

282 'Guid resolved to an object with no deep_url', dict(guid=guid) 

283 ) 

284 raise HTTPError(http_status.HTTP_404_NOT_FOUND) 

285 referent = guid_object.referent 

286 if referent is None: 

287 logger.error('Referent of GUID {0} not found'.format(guid)) 

288 raise HTTPError(http_status.HTTP_404_NOT_FOUND) 

289 if not referent.deep_url: 

290 raise HTTPError(http_status.HTTP_404_NOT_FOUND) 

291 

292 # Handle file `/download` shortcut with supported types. 

293 if suffix and suffix.rstrip('/').lower() == 'download': 

294 file_referent = None 

295 if isinstance(referent, Preprint) and referent.primary_file: 

296 file_referent = referent.primary_file 

297 elif isinstance(referent, BaseFileNode) and referent.is_file: 

298 file_referent = referent 

299 

300 if file_referent: 

301 if isinstance(file_referent.target, Preprint) and not file_referent.target.is_published: 

302 # TODO: Ideally, permissions wouldn't be checked here. 

303 # This is necessary to prevent a logical inconsistency with 

304 # the routing scheme - if a preprint is not published, only 

305 # admins and moderators should be able to know it exists. 

306 auth = Auth.from_kwargs(request.args.to_dict(), {}) 

307 # Check if user isn't a nonetype or that the user has admin/moderator/superuser permissions 

308 if auth.user is None or not (auth.user.has_perm('view_submissions', file_referent.target.provider) or 

309 file_referent.target.has_permission(auth.user, permissions.ADMIN)): 

310 raise HTTPError(http_status.HTTP_404_NOT_FOUND) 

311 

312 # Extend `request.args` adding `action=download`. 

313 request.args = request.args.copy() 

314 request.args.update({'action': 'download'}) 

315 # Do not include the `download` suffix in the url rebuild. 

316 url = _build_guid_url(unquote(file_referent.deep_url)) 

317 return proxy_url(url) 

318 

319 if suffix and suffix.rstrip('/').lower() == 'addtimestamp': 

320 file_referent = None 

321 if isinstance(referent, Preprint) and referent.primary_file: 

322 if not referent.is_published: 

323 # TODO: Ideally, permissions wouldn't be checked here. 

324 # This is necessary to prevent a logical inconsistency with 

325 # the routing scheme - if a preprint is not published, only 

326 # admins should be able to know it exists. 

327 auth = Auth.from_kwargs(request.args.to_dict(), {})

329 raise HTTPError(http_status.HTTP_404_NOT_FOUND)

332 file_referent = referent

336 request.args = request.args.copy()
