56 def _get_folders(self, library_id=None, folder_id=None): 

57 """ 

58 Get a list of a user's folders, either from their personal library, 

59 or a group library, if specified. 

60 

61 If folder_id is specified, will return the folders just within that folder. 

62 """ 

63 client = self._get_library(library_id) 

64 

65 # Note: Pagination is the only way to ensure all of the collections

68 if folder_id:

123 def _citations_for_folder(self, list_id, library_id=None): 

124 """Get all the citations in a specified collection 

125 

126 :param str list_id: ID for a Zotero collection. 

127 :return list of csljson objects representing documents. 

128 """ 

129 client = self._get_library(library_id) 

130 

131 citations = [] 

132 more = True 

133 offset = 0 

134 while more and len(citations) <= MAX_CITATION_LOAD: 

135 page = client.collection_items_top(list_id, content='csljson', limit=100, start=offset) 

136 citations = citations + page 

137 if len(page) == 0 or len(page) < 100: 

138 more = False 

139 else: 

140 offset = offset + len(page) 

141 return citations

143 def _citations_for_user(self, library_id=None): 

144 """Get all the citations from the user """ 

145 citations = [] 

146 more = True 

147 offset = 0 

148 client = self._get_library(library_id) 

149 

150 while more and len(citations) <= MAX_CITATION_LOAD: 

151 page = client.top(content='csljson', limit=100, start=offset) 

152 citations = citations + page 

153 if len(page) == 0 or len(page) < 100: 

154 more = False 

155 else: 

156 offset = offset + len(page) 

157 # Loops through all items in the library and extracts the keys for unfiled items 

158 unfiled_keys = [citation['data']['key'] for citation in client.top() if not citation['data']['collections']] 

159 

160 # Return only unfiled items in csljson format 

161 return [cite for cite in citations if cite['id'].split('/')[1] in unfiled_keys]

247 def get_top_level_folders(self, **kwargs): 

248 """ 

249 Returns serialized group libraries - your personal library along with any group libraries. 

250 

251 This is the top-tier of "folders" in Zotero. 

252 

253 You can use kwargs to refine what data is returned - how to limit the number of group libraries, 

254 whether to return the personal library alongside group_libraries, or append the total library count. 

255 """ 

256 # These kwargs are passed in from ZoteroViews > library_list 

257 limit = kwargs.get('limit', None) 

258 start = kwargs.get('start', None) 

259 return_count = kwargs.get('return_count', False) 

260 append_personal = kwargs.get('append_personal', True) 

261 try: 

262 # Fetch group libraries 

263 libraries = self.api._fetch_libraries(limit=limit, start=start) 

264 except zotero_errors.ResourceNotFound: 

265 raise HTTPError(404) 

266 except zotero_errors.UserNotAuthorised: 

267 raise HTTPError(403) 

268 except zotero_errors.HTTPError: 

269 sentry.log_exception() 

270 sentry.log_message('Unexpected Zotero Error when fetching group libraries.') 

271 raise HTTPError(500) 

272 

273 # Serialize libraries 

274 serialized = [] 

275 for library in libraries[:-1]: 

276 data = library['data'] 

277 serialized.append(self.serialize_folder('library', data['id'], data['name'], str(data['id']))) 

278 

279 if return_count: 

280 # Return total number of libraries as last item in list 

281 serialized.append(libraries[-1]) 

282 

283 if append_personal: 

284 # Append personal library as option alongside group libraries 

285 serialized.insert(0, self.serialize_folder('library', 'personal', 'My Library', 'personal')) 

286 return serialized

288 def get_sub_folders(self, library_id, folder_id=None, **kwargs): 

289 """ 

290 Returns serialized folders underneath a specific library/group - these are the lower tiers of folders in Zotero. 

291 

292 If no folder_id is specified, all folders in a flat manner are returned for the group library. 

293 If a folder_id is specified, only the subfolders within that folder are returned. 

294 """ 

295 try: 

296 sub_folders = self.api._get_folders(library_id=library_id, folder_id=folder_id) 

297 except zotero_errors.ResourceNotFound: 

298 raise HTTPError(404) 

299 except zotero_errors.UserNotAuthorised: 

300 raise HTTPError(403) 

301 except zotero_errors.HTTPError: 

302 sentry.log_exception() 

303 sentry.log_message('Unexpected Zotero Error when fetching folders.') 

304 raise HTTPError(500) 

305 

306 serialized = [] 

307 for folder in sub_folders: 

308 data = folder['data'] 

309 path = folder['library']['id'] if folder['library']['type'] == 'group' else 'personal' 

310 serialized.append(self.serialize_folder('folder', data['key'], data['name'], path, data['parentCollection'])) 

311 

312 if folder_id: 

313 return serialized 

314 else: 

315 all_documents = self.serialize_folder('folder', 'ROOT', 'All Documents', library_id, '__', None) 

316 return [all_documents] + serialized
