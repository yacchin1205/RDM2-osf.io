22def add_reviews_notification_setting(notification_type, state=None): 

23 if state: 

24 OSFUser = state.get_model('osf', 'OSFUser') 

25 NotificationSubscription = state.get_model('osf', 'NotificationSubscription') 

26 else: 

27 from osf.models import OSFUser, NotificationSubscription 

28 

29 active_users = OSFUser.objects.filter(date_confirmed__isnull=False).exclude(date_disabled__isnull=False).exclude(is_active=False).order_by('id') 

30 total_active_users = active_users.count() 

31 

32 logger.info('About to add a global_reviews setting for {} users.'.format(total_active_users)) 

33 

34 total_created = 0 

35 for user in active_users.iterator(): 

36 user_subscription_id = to_subscription_key(user._id, notification_type) 

37 

38 subscription = NotificationSubscription.load(user_subscription_id) 

39 if not subscription: 

40 logger.info('No {} subscription found for user {}. Subscribing...'.format(notification_type, user._id)) 

41 subscription = NotificationSubscription(_id=user_subscription_id, owner=user, event_name=notification_type) 

42 subscription.save() # Need to save in order to access m2m fields 

43 subscription.add_user_to_subscription(user, 'email_transactional') 

44 else: 

45 logger.info('User {} already has a {} subscription'.format(user._id, notification_type)) 

46 total_created += 1 

47 

48 logger.info('Added subscriptions for {}/{} users'.format(total_created, total_active_users))
