87def get_object_or_error(model_or_qs, query_or_pk=None, request=None, display_name=None): 

88 if not request: 

89 # for backwards compat with existing get_object_or_error usages 

90 raise TypeError('request is a required argument') 

91 

92 obj = query = None 

93 model_cls = model_or_qs 

94 select_for_update = check_select_for_update(request) 

95 

96 if isinstance(model_or_qs, QuerySet): 

97 # they passed a queryset 

98 model_cls = model_or_qs.model 

99 try: 

100 obj = model_or_qs.select_for_update().get() if select_for_update else model_or_qs.get() 

101 except model_cls.DoesNotExist: 

102 raise NotFound 

103 

104 elif isinstance(query_or_pk, basestring): 

105 # they passed a 5-char guid as a string 

106 if issubclass(model_cls, GuidMixin): 

107 # if it's a subclass of GuidMixin we know it's primary_identifier_name 

108 query = {'guids___id': query_or_pk} 

109 else: 

110 if hasattr(model_cls, 'primary_identifier_name'): 

111 # primary_identifier_name gives us the natural key for the model 

112 query = {model_cls.primary_identifier_name: query_or_pk} 

113 else: 

114 # fall back to modmcompatiblity's load method since we don't know their PIN 

115 obj = model_cls.load(query_or_pk, select_for_update=select_for_update) 

116 else: 

117 # they passed a query 

118 try: 

119 obj = model_cls.objects.filter(query_or_pk).select_for_update().get() if select_for_update else model_cls.objects.get(query_or_pk) 

120 except model_cls.DoesNotExist: 

121 raise NotFound 

122 

123 if not obj: 

124 if not query: 

125 # if we don't have a query or an object throw 404 

126 raise NotFound 

127 try: 

128 # TODO This could be added onto with eager on the queryset and the embedded fields of the api 

129 if isinstance(query, dict): 

130 obj = model_cls.objects.get(**query) if not select_for_update else model_cls.objects.filter(**query).select_for_update().get() 

131 else: 

132 obj = model_cls.objects.get(query) if not select_for_update else model_cls.objects.filter(query).select_for_update().get() 

133 except ObjectDoesNotExist: 

134 raise NotFound 

135 

136 # For objects that have been disabled (is_active is False), return a 410. 

137 # The User model is an exception because we still want to allow 

138 # users who are unconfirmed or unregistered, but not users who have been 

139 # disabled. 

140 if model_cls is OSFUser and obj.is_disabled: 

141 raise UserGone(user=obj) 

142 elif model_cls is not OSFUser and not getattr(obj, 'is_active', True) or getattr(obj, 'is_deleted', False) or getattr(obj, 'deleted', False): 

143 if display_name is None: 

144 raise Gone 

145 else: 

146 raise Gone(detail='The requested {name} is no longer available.'.format(name=display_name)) 

147 return obj

168def extend_querystring_params(url, params): 

169 scheme, netloc, path, query, _ = urlsplit(url) 

170 orig_params = parse_qs(query) 

171 orig_params.update(params) 

172 query = urlencode(orig_params, True) 

173 return urlunsplit([scheme, netloc, path, query, ''])

181def has_admin_scope(request): 

182 """ Helper function to determine if a request should be treated 

183 as though it has the `osf.admin` scope. This includes both 

184 tokened requests that do, and requests that are made via the 

185 OSF (i.e. have an osf cookie) 

186 """ 

187 cookie = request.COOKIES.get(website_settings.COOKIE_NAME) 

188 if cookie: 

189 return bool(get_session_from_cookie(cookie)) 

190 

191 token = request.auth 

192 if token is None or not isinstance(token, CasResponse): 

193 return False 

194 

195 return set(ComposedScopes.ADMIN_LEVEL).issubset(normalize_scopes(token.attributes['accessTokenScope']))

256def check_user_can_create_project(user): 

257 """Check if user can create project. 

258 :param user: OSFUser 

259 :return: True if user can create project, False otherwise 

260 """ 

261 from osf.models import ProjectLimitNumberSetting, ProjectLimitNumberSettingAttribute, \ 

262 AbstractNode, ProjectLimitNumberDefault 

263 

264 # If not user return false 

265 if not user: 

266 return False 

267 

268 # if user has no affiliated institution return true 

269 institution = user.affiliated_institutions.first() 

270 if not institution: 

271 return True 

272 

273 # Get the list setting for institution 

274 setting_list = ProjectLimitNumberSetting.objects.filter( 

275 institution_id=institution.id, 

276 is_availability=True, 

277 is_deleted=False, 

278 ).order_by('priority').all() 

279 setting_id_list = [s.id for s in setting_list] 

280 # Get setting list attribute by setting 

281 all_setting_attribute_list = (ProjectLimitNumberSettingAttribute.objects.select_related( 

282 'attribute', 

283 ).filter( 

284 setting_id__in=setting_id_list, 

285 is_deleted=False, 

286 ).annotate( 

287 setting_type=F('attribute__setting_type'), 

288 attribute_name=F('attribute__attribute_name'), 

289 setting_id=F('setting_id'), 

290 ).order_by('id').values( 

291 'id', 

292 'attribute_name', 

293 'setting_type', 

294 'attribute_value', 

295 'setting_id', 

296 )) 

297 project_limit_number = None 

298 user_dict = user.__dict__ 

299 

300 # Map attribute list to dictionary 

301 setting_attributes_dict = {} 

302 for setting_attribute in all_setting_attribute_list: 

303 setting_id = setting_attribute.get('setting_id') 

304 setting_attributes_dict.setdefault(setting_id, []).append(setting_attribute) 

305 

306 user_extended_data = UserExtendedData.objects.filter(user_id=user_dict.get('id')).first() 

307 # Check user by setting list attribute condition for each setting 

308 for setting in setting_list: 

309 if check_logic_condition(user_dict, setting_attributes_dict.get(setting.id, []), user_extended_data): 

310 project_limit_number = setting.project_limit_number 

311 break 

312 

313 # If no setting found, get default or use no limit (-1) 

314 if project_limit_number is None: 

315 default_limit = ProjectLimitNumberDefault.objects.filter( 

316 institution_id=institution.id, 

317 ).values_list('project_limit_number', flat=True).first() 

318 if default_limit is not None: 

319 project_limit_number = default_limit 

320 else: 

321 project_limit_number = NO_LIMIT 

322 

323 # Return if no limit 

324 if project_limit_number == NO_LIMIT: 

325 return True 

326 

327 # Get number of created project 

328 created_project_number = AbstractNode.objects.filter( 

329 type='osf.node', 

330 creator_id=user.id, 

331 is_deleted=False, 

332 ).count() 

333 

334 return project_limit_number > created_project_number
