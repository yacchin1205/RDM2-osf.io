232 def get_user_quota_type_for_nii_storage(self): 

233 """If institution is using NII Storage, return user quota type. 

234 Otherwise, return None 

235 """ 

236 from addons.osfstorage.models import Region 

237 from osf.models import UserQuota 

238 region = Region.objects.filter(_id=self._id).first() 

239 if not region: 

240 # Institution does not have its own institutional storage, return NII_STORAGE 

241 return UserQuota.NII_STORAGE 

242 if region.waterbutler_settings.get('storage', {}).get('type') == Region.NII_STORAGE: 

243 # Institutional storage is using NII Storage, return CUSTOM_STORAGE 

244 return UserQuota.CUSTOM_STORAGE 

245 else: 

246 # Institutional storage is not using NII Storage, return None 

247 return None
