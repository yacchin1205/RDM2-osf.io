43 def send_mail(self): 

44 """ 

45 Grabs the data from this email, checks for user subscription to help mails, 

46 

47 constructs the mail object and checks presend. Then attempts to send the email 

48 through send_mail() 

49 :return: boolean based on whether email was sent. 

50 """ 

51 mail_struct = queue_mail_types[self.email_type] 

52 presend = mail_struct['presend'](self) 

53 mail = Mail( 

54 mail_struct['template'], 

55 subject=mail_struct['subject'], 

56 categories=mail_struct.get('categories', None) 

57 ) 

58 self.data['osf_url'] = osf_settings.DOMAIN 

59 if presend and self.user.is_active and self.user.osf_mailing_lists.get(osf_settings.OSF_HELP_LIST): 

60 send_mail(self.to_addr or self.user.username, mail, mimetype='html', **(self.data or {})) 

61 self.sent_at = timezone.now() 

62 self.save() 

63 return True 

64 else: 

65 self.__class__.delete(self) 

66 return False

68 def find_sent_of_same_type_and_user(self): 

69 """

71 Does not look for queue-up emails. 

72 :return: a list of those emails

77def queue_mail(to_addr, mail, send_at, user, **context): 

78 """ 

79 Queue an email to be sent using send_mail after a specified amount 

80 of time and if the presend returns True. The presend is attached to 

81 the template under mail. 

82 

83 :param to_addr: the address email is to be sent to 

84 :param mail: the type of mail. Struct following template: 

85 { 'presend': function(), 

86 'template': mako template name, 

87 'subject': mail subject } 

88 :param send_at: datetime object of when to send mail 

89 :param user: user object attached to mail 

90 :param context: IMPORTANT kwargs to be attached to template. 

91 Sending mail will fail if needed for template kwargs are 

92 not parameters. 

93 :return: the QueuedMail object created 

94 """ 

95 if waffle.switch_is_active(features.DISABLE_ENGAGEMENT_EMAILS) and mail.get('engagement', False): 

96 return False 

97 new_mail = QueuedMail( 

98 user=user, 

99 to_addr=to_addr, 

100 send_at=send_at, 

101 email_type=mail['template'], 

102 data=context 

103 ) 

104 new_mail.save() 

105 return new_mail
